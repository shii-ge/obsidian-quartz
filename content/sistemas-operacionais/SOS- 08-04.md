# Funções em shell

- Exemplo de função em shell.

```sh
#!/bin/sh

sim_ou_nao() {
	echo "Seu nome é $*?"
	while true
		do
			echo -n "Responder sim ou nao: "
			read x
			case "$x" in
				s | sim ) return 0;;
				n | nao ) return 1;;
				* ) echo "Resposta errada."
			esac
		done
}

echo "Parametros: $*"
if sim_ou_nao "$1"
	then
		echo "Ola $1, belo nome."
	else
		echo "Deixa pra lá."
	fi
exit 0
```
